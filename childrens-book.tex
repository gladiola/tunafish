\newcommand{\illustrationplaceholder}[1]{%
  \setlength{\fboxsep}{0pt}%
  \colorlet{placeholderborder}{black!40}%
  \colorlet{placeholderfill}{black!4}%
  \fcolorbox{placeholderborder}{placeholderfill}{%
    \begin{minipage}[c][\illustrationboxheight][c]{\illustrationboxwidth}
      \centering
      {\headingfont\fontsize{22}{26}\selectfont Illustration Placeholder\par}
      \vspace{1em}
      {\Large #1\par}
      \vspace{1em}
      {\normalsize Add an artwork path as the third \texttt{\string\storyspread} argument.\par}
    \end{minipage}%
  }%
}

\newcommand{\storypage}[2]{%
  \global\storypageoktrue
  \thispagestyle{empty}%
  \vspace*{\storytopoffset}%
  \sbox{\storybox}{%
    \begin{minipage}[t]{\textwidth}
      {\chapterfont\fontsize{26}{30}\selectfont #1\par}%
      \vspace{1.75em}%
      {\raggedright\fontsize{18}{26}\selectfont #2\par}%
    \end{minipage}%
  }%
  \ifdim\dimexpr\ht\storybox+\dp\storybox\relax>\storyboxheight
    \global\storypageokfalse
    \PackageError{childrens-book}{Story page '#1' is too long for a single spread page}{Split the text across multiple \string\storyspread\space commands or shorten the copy.}%
  \else
    \noindent\parbox[t][\storyboxheight][t]{\textwidth}{\usebox{\storybox}}%
    \vfill
    \newpage
  \fi
}

\newcommand{\illustrationpage}[2]{%
  \thispagestyle{empty}%
  \vspace*{\fill}%
  \begin{center}
    \ifstrempty{#2}
      {\illustrationplaceholder{#1}}
      {%
        \IfFileExists{#2}
          {\includegraphics[width=\illustrationboxwidth,height=\illustrationboxheight,keepaspectratio]{#2}}
          {%
            \PackageWarning{childrens-book}{Illustration file '#2' was not found. Using the placeholder panel instead}%
            \illustrationplaceholder{#1}%
          }%
      }
  \end{center}
  \vspace*{\fill}%
  \newpage
}

\NewDocumentCommand{\storyspread}{m m m}{%
  \storypage{#1}{#2}
  \ifstorypageok
    \illustrationpage{#1}{#3}
  \fi
}
